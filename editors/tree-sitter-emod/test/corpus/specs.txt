========================
Two spec blocks in one slice
========================

context "Lending" {
  aggregate "Loan" {
    slice "Borrow a Copy" {
      spec "borrows a free copy" {
        given []
        when BorrowCopy
        then [CopyBorrowed]
      }
      spec "rejects a second borrow" {
        given [CopyBorrowed]
        when BorrowCopy
        then rejected OneCopyPerLoan
      }
    }
  }
}

========================
A spec before and after a flow
========================

context "Lending" {
  aggregate "Loan" {
    slice "Borrow a Copy" {
      spec "before the flow" {
        when BorrowCopy
        then [CopyBorrowed]
      }
      flow {
        command -> event: BorrowCopy -> CopyBorrowed
      }
      spec "after the flow" {
        when BorrowCopy
        then [CopyBorrowed]
      }
    }
  }
}
